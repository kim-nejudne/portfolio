## Concept

A function over a union handles one member per branch. Inside each branch the parameter has that member's type, so its properties are available without a cast.

```ts twoslash
type Command = { type: 'move'; dx: number } | { type: 'say'; text: string };

function run(command: Command): string {
  if (command.type === 'say') {
    return command.text;
    //             ^?
  }
  return `moved by ${command.dx}`;
}
```

## Examples

The same elimination works for `undefined`. A variable typed `Shape | undefined` starts empty; checking it against `undefined` narrows it to `Shape` for the rest of the branch:

```ts run
type Shape =
  { kind: 'circle'; radius: number } | { kind: 'square'; side: number };

function area(shape: Shape): number {
  return shape.kind === 'circle'
    ? Math.PI * shape.radius ** 2
    : shape.side ** 2;
}

function largest(shapes: readonly Shape[]): Shape | undefined {
  let best: Shape | undefined;
  for (const shape of shapes) {
    if (best === undefined || area(shape) > area(best)) {
      best = shape;
    }
  }
  return best;
}

console.log(
  largest([
    { kind: 'circle', radius: 1 },
    { kind: 'square', side: 3 },
  ]),
);
console.log(largest([]));
```

## Pitfalls

**Using the value before narrowing it.** `best` may still be `undefined` when the loop starts, and the compiler says so:

```ts run errors
type Shape =
  { kind: 'circle'; radius: number } | { kind: 'square'; side: number };

function first(shapes: readonly Shape[]): string {
  let best: Shape | undefined;
  for (const shape of shapes) {
    best = shape;
    break;
  }
  return best.kind;
}
```
